455void ApplicationInterface::setMemento(Memento* memento) {
456 if (memento == nullptr || memento->getRoot() == nullptr) {
457 return;
458 }
459 switch (memento->getType()) {
460 case MementoType::company:
461 {
462 CompanyNode* restored = ((CompanyNode*)memento->getRoot())->clone();
463 delete companyRoot;
464 companyRoot = restored;
465 assignedTeams.clear();
466 for (const auto& entry : assignments) {
467 for (Team* team : getTeams()) {
468 if (team->getName() == entry.second) {
469 assignedTeams[entry.first] = team;
470 break;
471 }
472 }
473 }
474 break;
475 }
476 case MementoType::workflow:
477 {
478 WorkNode* restored = ((WorkNode*)memento->getRoot())->clone();
479 delete workflowRoot;
480 workflowRoot = restored;
481 assignments.clear();
482 assignedTeams.clear();
483 refreshWorkflow();
484 break;
485 }
486 default:
487 break;
488 }
489}

595bool ApplicationInterface::setCompetency(Team* team, int memberIndex, const std::string& competency) {
596 if (team == nullptr || EmployeeCompetency::getCompetencyLevel(competency) < 0) {
597 return false;
598 }
599 EmployeeComponent* old = team->getMember(memberIndex);
600 if (old == nullptr) {
601 return false;
602 }
603 std::string profession = team->getProfession();
604 if (profession.empty()) {
605 for (const std::string& p : getProfessions()) {
606 if (old->compareProfession(p)) {
607 profession = p;
608 break;
609 }
610 }
611 }
612 ConcreteEmployeeBuilder builder;
613 builder.reset();
614 if (!profession.empty()) {
615 builder.addProfession(profession);
616 }
617 builder.addCompetency(competency);
618 EmployeeComponent* replacement = builder.finish(old->getName(), old->getCore()->getKind() == "Manager");
619 if (!team->replaceMember(memberIndex, replacement)) {
620 delete replacement;
621 return false;
622 }
623 delete old;
624 return true;

721bool ApplicationInterface::renameWorkNode(WorkNode* node, const std::string& name) {
722 if (node == nullptr || name.empty()) {
723 return false;
724 }
725 if (node->getName() == name) {
726 return true;
727 }
728 if (getWorkflow()->hasName(name)) {
729 return false;
730 }
731 std::string oldName = node->getName();
732 node->setName(name);
733 for (WorkNode* other : getWorkNodes()) {
734 const std::vector<std::string> prereqs = other->getPrerequisites();
735 if (std::find(prereqs.begin(), prereqs.end(), oldName) != prereqs.end()) {
736 other->removePrerequisite(oldName);
737 other->addPrerequisite(name);
738 }
739 }
740 auto it = assignments.find(oldName);
741 if (it != assignments.end()) {
742 assignments[name] = it->second;
743 assignments.erase(oldName);
744 }
745 auto teamIt = assignedTeams.find(oldName);
746 if (teamIt != assignedTeams.end()) {
747 assignedTeams[name] = teamIt->second;
748 assignedTeams.erase(oldName);
749 }
750 return true;
751}

753// for gui:
754bool ApplicationInterface::removeWorkNode(WorkNode* node) {
755 if (node == nullptr || node == workflowRoot || node->getParent() == nullptr) {
756 return false;
757 }
758 std::vector<WorkNode*> doomed;
759 WorkIterator doomedIt(node, false);
760 for (doomedIt.first(); !doomedIt.isDone(); doomedIt.next()) {
761 doomed.push_back(doomedIt.currentItem());
762 }
763
764 node->getParent()->remove(node);
765
766 std::vector<WorkNode*> released;
767 for (WorkNode* gone : doomed) {
768 std::string goneName = gone->getName();
769 assignments.erase(goneName);
770 assignedTeams.erase(goneName);
771 for (WorkNode* other : getWorkNodes()) {
772 const std::vector<std::string>& prereqs = other->getPrerequisites();
773 if (std::find(prereqs.begin(), prereqs.end(), goneName) != prereqs.end()) {
774 other->removePrerequisite(goneName);
775 released.push_back(other);
776 }
777 }
778 }
779 delete node;
780 refreshWorkflow();
781 for (WorkNode* other : released) {
782 if (other->getPrerequisites().empty() && other->getStateName() == "WaitingForPrerequisites") {
783 other->advanceState();
784 }
785 }
786 return true;
787}

816bool ApplicationInterface::addPrerequisite(WorkNode* node, const std::string& name) {
817 if (node == nullptr || name.empty() || node->getName() == name) {
818 return false;
819 }
820 WorkNode* target = findWorkNode(name);
821 if (target == nullptr || target->isCompleted()) {
822 return false;
823 }
824 const std::vector<std::string>& existing = node->getPrerequisites();
825 if (std::find(existing.begin(), existing.end(), name) != existing.end()) {
826 return false;
827 }
828 for (WorkNode* up = node->getParent(); up != nullptr; up = up->getParent()) {
829 if (up == target) {
830 return false;
831 }
832 }
833 if (dependsOn(target, node->getName(), 0)) {
834 return false;
835 }
836 node->addPrerequisite(name);
837 if (node->isReady()) {
838 node->setState("WaitingForPrerequisites");
839 }
840 return true;
841}

861 if (unit == nullptr || team == nullptr) {
862 return "Nothing to assign.";
863 }
864 std::string state = unit->getStateName();
865 if (state == "Completed" || state == "Cancelled" || state == "Failed") {

868 std::string required = unit->getRequiredProfession();

870 return unit->getName() + " needs a " + required + " team, not " + team->getName() + ".";

873 return "Nobody in the company has the profession " + required + ".";
874 }
875 if (state == "WaitingForPrerequisites") {
876 if (unit->isWaitingForPrerequisites()) {
877 return unit->getName() + " is still waiting for its prerequisites.";
878 }
879 unit->advanceState();
880 }
881 return "";

985 if (unit == nullptr) {
986 return "Nothing to change.";
987 }
988 const char* known[] = { "WaitingForPrerequisites", "Ready", "Ongoing", "Escalated", "Completed", "Cancelled", "Failed" };
989 bool found = false;

992 found = true;
993 }

995 if (!found) {
996 return stateName + " is not a state.";
997 }
998 if (unit->getStateName() == stateName) {

1000 }
1001 if (stateName == "Completed") {
1002 unit->setState("Escalated");
1003 unit->advanceState();
1004 }
1005 else {
1006 unit->setState(stateName);
1007 }
1008 return unit->getName() + " is now " + unit->getStateName() + ".";
