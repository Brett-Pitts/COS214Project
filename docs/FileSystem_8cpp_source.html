20// Reads one line and strips a trailing '\r' so files saved on Windows load on Linux.
21static bool readLine(std::istream& file, std::string& line) {
22 if (!std::getline(file, line)) {
23 return false;
24 }
25 if (!line.empty() && line[line.size() - 1] == '\r') {
26 line.erase(line.size() - 1);
27 }
28 return true;
29}

40static void readPrerequisites(std::istream& file, WorkNode* node) {

54// Reads one decorator or core part of an employee (the line after the tag is the
55// name, followed by the matching End line). Returns nullptr for an unknown tag.
56static EmployeeComponent* readEmployeePart(std::istream& file, const std::string& tag) {
57 std::string value;
58 std::string end;
59 EmployeeComponent* part = nullptr;
60
61 if (tag != "Profession" && tag != "Competency" && tag != "Employee" && tag != "Manager") {
62 return nullptr;
63 }
64 if (!readLine(file, value)) {
65 return nullptr;
66 }
67
68 if (tag == "Profession") {
69 part = new EmployeeProfession(value);
70 }
71 else if (tag == "Competency") {
72 part = new EmployeeCompetency(value);
73 }
74 else if (tag == "Employee") {
75 part = new Employee(value);
76 }
77 else {
78 part = new Manager(value);
79 }
80
81 if (!readLine(file, end) || end != "End" + tag) {
82 std::cerr << "Error reading file, End" << tag << " not found\n";
83 }
84 return part;
85}

120void FileSystem::saveCompany(std::ofstream& file, CompanyNode* root) {
121 if (root == nullptr) {
122 return;
123 }
124
125 Team* team = dynamic_cast<Team*>(root);
126 if (team != nullptr) {
127 file << "Team\n";
128 file << team->profession << '\n';
129
130 for (EmployeeComponent* emp : team->members) {
131 file << "Member\n";
132
133 // Walk from the outermost decorator down to the core, writing each part.
134 EmployeeComponent* current = emp;
135 while (current != nullptr) {
136 std::string kind = current->getKind();
137
138 if (kind == "Profession") {
139 file << "Profession\n";
140 file << static_cast<EmployeeProfession*>(current)->professionName << '\n';
141 file << "EndProfession\n";
142 current = static_cast<EmployeeDetail*>(current)->wrapped;
143 }
144 else if (kind == "Competency") {
145 file << "Competency\n";
146 file << static_cast<EmployeeCompetency*>(current)->competencyName << '\n';
147 file << "EndCompetency\n";
148 current = static_cast<EmployeeDetail*>(current)->wrapped;
149 }
150 else if (kind == "Employee") {
151 file << "Employee\n";
152 file << current->getName() << '\n';
153 file << "EndEmployee\n";
154 break;
155 }
156 else if (kind == "Manager") {
157 file << "Manager\n";
158 file << current->getName() << '\n';
159 file << "EndManager\n";
160 break;
161 }
162 else {
163 break;
164 }
165 }
166
167 file << "EndMember\n";
168 }
169 file << "EndTeam\n";
170 }
171 else {
172 // Department (this also handles an empty department correctly)
173 file << "Department\n";
174 for (int i = 0;; i++) {
175 CompanyNode* child = root->getChild(i);
176 if (child == nullptr) {
177 break;
178 }
179 saveCompany(file, child);
180 }
181 file << "EndDepartment\n";
182 }

265void FileSystem::loadCompany(std::ifstream& file, CompanyNode*& root) {
266 std::string line;
267 if (!readLine(file, line)) {
268 return;
269 }
270
271 if (line == "EndDepartment") {
272 return;
273 }
274
275 if (line == "Department") {
276 Department* department = new Department();
277
278 while (true) {
279 CompanyNode* child = nullptr;
280 loadCompany(file, child);
281 if (child == nullptr) {
282 break;
283 }
284 department->add(child);
285 }
286 root = department;
287 }
288 else if (line == "Team") {
289
290 std::string profession;
291 std::streampos position = file.tellg();
292 if (!readLine(file, profession)) {
293 return;
294 }
295 if (profession == "Member" || profession == "EndTeam") {
296 file.clear();
297 file.seekg(position);
298 profession = "";
299 }
300
301 Team* team = new Team(profession);
302
303 while (readLine(file, line)) {
304 if (line == "EndTeam") {
305 break;
306 }
307
308 if (line == "Member") {
309 EmployeeComponent* member = nullptr;
310
311 while (readLine(file, line)) {
312 if (line == "EndMember") {
313 break;
314 }
315 EmployeeComponent* part = readEmployeePart(file, line);
316 if (part == nullptr) {
317 continue;
318 }
319 if (member == nullptr) {
320 member = part;
321 }
322 else {
323 member->wrap(part);
324 }
325 }
326
327 if (member != nullptr && !team->addMember(member)) {
328 std::cerr << "Could not add member " << member->getName() << " to the team\n";
329 delete member;
330 }
331 }
332 }
333 root = team;
334 }

337void FileSystem::loadWorkflow(std::ifstream& file, WorkNode*& root) {
338
339 std::string line;
340 if (!readLine(file, line)) {
341 return;
342 }
343
344 if (line == "EndProject" || line == "EndChildren") {
345 return;
346 }
347
348 if (line == "Project") {
349 std::string name;
350 std::string maxRetries;
351 std::string stateName;
352 if (!readLine(file, name) || !readLine(file, maxRetries) || !readLine(file, stateName)) {
353 std::cerr << "Error reading file, project header incomplete\n";
354 return;
355 }
356
357 int retries = 0;
358 try {
359 retries = std::stoi(maxRetries);
360 }
361 catch (const std::exception&) {
362 std::cerr << "Error reading file, bad retry count: " << maxRetries << '\n';
363 return;
364 }
365
366 Project* project = new Project(retries, name);
367 project->setState(stateName);
368 readPrerequisites(file, project);
369
370 if (readLine(file, line) && line == "Children") {
371 while (true) {
372 WorkNode* child = nullptr;
373 loadWorkflow(file, child);
374 if (child == nullptr) {
375 break; // "EndChildren" (consumed above) or an error
376 }
377 project->add(child);
378 }
379
380 // Consume this project's own end marker so the parent's loop stays in step.
381 if (!readLine(file, line) || line != "EndProject") {
382 std::cerr << "Error reading file, EndProject not found!\n";
383 }
384 }
385 else {
386 std::cerr << "Error reading file, Children not found!\n";
387 }
388 root = project;
389 }
390
391 else if (line == "Unit") {
392
393 std::string name;
394 std::string required;
395 if (!readLine(file, name) || !readLine(file, required)) {
396 std::cerr << "Error reading file, unit incomplete\n";
397 return;
398 }
399
400 WorkUnit* unit = new WorkUnit(name, required);
401
402 // Newer files store the state and prerequisites; older ones go straight to EndUnit.
403 if (!readLine(file, line)) {
404 std::cerr << "Error reading file, EndUnit not found!\n";
405 }
406 else if (line != "EndUnit") {
407 unit->setState(line);
408 readPrerequisites(file, unit);
409 if (!readLine(file, line) || line != "EndUnit") {
410 std::cerr << "Error reading file, EndUnit not found!\n";
411 }
412 }
413 root = unit;
414 }
