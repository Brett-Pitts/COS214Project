37static EmployeeComponent* makeEmployee(const std::string& name,
38 const std::string& profession,
39 const std::string& competency,
40 bool isManager = false) {
41 EmployeeComponent* core;
42 if (isManager) {
43 core = new Manager(name);
44 }
45 else {
46 core = new Employee(name);
47 }
48
49 EmployeeComponent* prof = new EmployeeProfession(profession);
50 prof->wrap(core);
51
52 EmployeeComponent* comp = new EmployeeCompetency(competency);
53 comp->wrap(prof);
54
55 return comp;
56}
